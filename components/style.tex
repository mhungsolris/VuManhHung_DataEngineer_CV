% Render a left label and right-aligned date on one line when they fit. Long
% labels fall back to a separate right-aligned date line without overlap.
\newlength{\cvlabelwidth}
\newlength{\cvdatewidth}
\newcommand{\cvdatedline}[2]{%
  \settowidth{\cvlabelwidth}{#1}%
  \settowidth{\cvdatewidth}{#2}%
  \ifdim\dimexpr\cvlabelwidth+\cvdatewidth+1.5em\relax>\linewidth
    #1\par\nopagebreak
    \hfill #2\par\nopagebreak
  \else
    #1\hfill #2\par\nopagebreak
  \fi
}
